#include <iostream>
#include <string>
#include <map>
using namespace std;

int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    string str;
    cin >> str;

    map<char, int> m;
    for (char c : str)
    {
        if (c >= 'a' && c <= 'z')
            c = c - 'a' + 'A';
        m[c]++;
    }

    int maxVal = 0;
    char maxChar = '?';
    bool dup = false;

    for (auto& [key, val] : m)
    {
        if (val > maxVal)        { maxVal = val; maxChar = key; dup = false; }
        else if (val == maxVal)  { dup = true; }
    }

    cout << (dup ? '?' : maxChar) << "\n";
    return 0;
}
