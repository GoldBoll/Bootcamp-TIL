// 백준 10828
// https://www.acmicpc.net/problem/10828


#include <iostream>
#include <stack>
#include <string>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    int n;
    cin >> n;

    stack<int> s;
    string cmd;

    for(int i=0; i<n; i++){
        cin >> cmd;
        if(cmd == "push"){
            int num;
            cin >> num;
            s.push(num);
        }
        else if(cmd == "pop"){
            if(s.empty())
                cout << "-1\n";
            else{
                cout << s.top() << "\n";
                s.pop();
            }
        }
        else if(cmd == "size"){
            cout << s.size() << "\n";
        }
        else if(cmd == "empty"){
            cout << s.empty() << "\n";
        }
        else if(cmd == "top"){
            if(s.empty())
                cout << "-1\n";
            else
                cout << s.top() << "\n";
        }
    }

    return 0;
}
