// LeetCode 88 - Merge Sorted Array (Easy)
// https://leetcode.com/problems/merge-sorted-array/

// 문제 설명
// 오름차순 정렬된 두 배열 nums1(m개), nums2(n개)를 하나의 배열로 병합
// 결과는 nums1 내부에 저장해야 함
// nums1의 크기는 m+n이며, 뒤쪽 n개는 0으로 채워진 빈 공간

// 입력
// nums1 = [1,2,3,0,0,0], m = 3, nums2 = [2,5,6], n = 3

// 출력
// nums1 = [1,2,2,3,5,6]

#include <iostream>
#include <vector>
using namespace std;

// 내 풀이 1: 임시 배열 사용 (O(m+n) 시간, O(m+n) 공간)
// 두 포인터로 앞에서부터 비교하며 임시 배열에 채운 뒤 nums1에 복사
void merge_v1(vector<int>& nums1, int m, vector<int>& nums2, int n) {
    vector<int> solution;
    int count1 = 0, count2 = 0;

    for (int i = 0; i < m + n; i++) {
        if (count1 >= m)
            solution.push_back(nums2[count2++]);
        else if (count2 >= n)
            solution.push_back(nums1[count1++]);
        else if (nums1[count1] <= nums2[count2])
            solution.push_back(nums1[count1++]);
        else
            solution.push_back(nums2[count2++]);
    }

    nums1 = solution;
}

//추천 풀이 2: 뒤에서부터 채우기 (O(m+n) 시간, O(1) 공간)
// nums1 뒤쪽 빈 공간을 활용해 큰 값부터 역순으로 삽입
// 앞에서 채우면 nums1 원소를 덮어쓰는 문제가 생기므로 뒤에서 채움
void merge_v2(vector<int>& nums1, int m, vector<int>& nums2, int n) {
    int p1 = m - 1;      // nums1 유효 원소 끝
    int p2 = n - 1;      // nums2 끝
    int p  = m + n - 1;  // 채울 위치 (뒤부터)

    while (p2 >= 0) {
        if (p1 >= 0 && nums1[p1] > nums2[p2])
            nums1[p--] = nums1[p1--];
        else
            nums1[p--] = nums2[p2--];
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    vector<int> nums1 = {1, 2, 3, 0, 0, 0};
    vector<int> nums2 = {2, 5, 6};

    merge_v2(nums1, 3, nums2, 3);

    for (int x : nums1)
        cout << x << " ";
    cout << "\n";

    return 0;
}
